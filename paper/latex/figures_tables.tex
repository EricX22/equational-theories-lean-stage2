\begin{figure*}[t]
\centering
\begin{tikzpicture}[
  >={Stealth[width=2mm,length=2.4mm]},
  every node/.style={font=\small},
  box/.style={draw, rounded corners, align=center, inner sep=5pt,
              minimum height=10mm, text width=26mm},
  data/.style={draw, align=center, inner sep=5pt, minimum height=10mm,
               text width=18mm},
  node distance=9mm and 12mm
]
  \node[data]                         (law)    {Law $L:\ x=T$};
  \node[box, right=of law]            (adm)    {Admissibility prover\\[-1pt]
                                                {\footnotesize no nontrivial\\[-2pt]
                                                 finite model}};
  \node[box, right=of adm]            (task)   {Two-sided task\\[-1pt]
                                                {\footnotesize (exactly one holds)}};
  \node[box, above right=of task]     (aust)   {Construct model\\[-1pt]
                                                $\Rightarrow$ \textsc{AustinGoal}};
  \node[box, below right=of task]     (triv)   {Prove $L\models x{=}y$\\[-1pt]
                                                $\Rightarrow$ \textsc{TrivialGoal}};
  \node[box, right=34mm of task]      (judge)  {Lean judge\\[-1pt]
                                                {\footnotesize kernel $+$\\[-2pt]
                                                 axiom allowlist}};
  \node[data, right=of judge]         (out)    {Accept /\\ Reject};

  \draw[->] (law)  -- (adm);
  \draw[->] (adm)  -- (task);
  \draw[->] (task) -- (aust);
  \draw[->] (task) -- (triv);
  \draw[->] (aust) -- (judge);
  \draw[->] (triv) -- (judge);
  \draw[->] (judge) -- (out);
\end{tikzpicture}
\caption{The benchmark pipeline. Each generated law carries a machine-checked
certificate of no nontrivial finite model (admissibility); by the dichotomy of
\Cref{sec:task} exactly one of the two branches is achievable. A submitted answer is
a Lean proof of the corresponding goal, accepted only if the kernel checks it and its
axiom footprint lies within a fixed allowlist. The judge is agnostic to the model's
form.}
\label{fig:pipeline}
\end{figure*}
