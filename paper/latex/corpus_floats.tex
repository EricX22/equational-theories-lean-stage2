\begin{figure}[t]
\centering
\begin{tikzpicture}[
  >={Stealth[width=1.8mm,length=2.2mm]},
  font=\small,
  proc/.style={draw, rounded corners, align=center, inner sep=4pt,
               minimum height=8mm, text width=34mm},
  dec/.style={draw, diamond, aspect=2, align=center, inner sep=1pt,
              text width=26mm},
  term/.style={draw, dashed, rounded corners, align=center, inner sep=4pt,
               minimum height=7mm, text width=20mm},
  node distance=7mm and 12mm
]
  \node[proc]                     (seed)   {Seed Austin laws};
  \node[proc, below=of seed]      (ext)    {Extend by one operation};
  \node[dec,  below=of ext]       (triv)   {Provably trivial?};
  \node[term, right=18mm of triv] (prune)  {Prune};
  \node[dec,  below=of triv]      (adm)    {No nontrivial finite model, certified?};
  \node[term, right=18mm of adm]  (disc)   {Discard};
  \node[proc, below=of adm]       (dedup)  {Deduplicate against seeds};
  \node[proc, below=of dedup]     (corpus) {Corpus (hard tier $+$ resolved)};

  \draw[->] (seed) -- (ext);
  \draw[->] (ext)  -- (triv);
  \draw[->] (triv) -- node[above,font=\footnotesize]{yes} (prune);
  \draw[->] (triv) -- node[right,font=\footnotesize]{no}  (adm);
  \draw[->] (adm)  -- node[above,font=\footnotesize]{no}  (disc);
  \draw[->] (adm)  -- node[right,font=\footnotesize]{yes} (dedup);
  \draw[->] (dedup) -- (corpus);
\end{tikzpicture}
\caption{The construction and pruning pipeline. Candidates are grown by one-operation
extension, then screened cheap-to-expensive: laws a fast proof search settles as trivial
are pruned, and the rest are admitted only on a machine-checked proof that they have no
nontrivial finite model---any law that fails to earn this certificate is discarded.
Admitted laws are deduplicated against their seeds before entering the corpus.}
\label{fig:construction}
\end{figure}
